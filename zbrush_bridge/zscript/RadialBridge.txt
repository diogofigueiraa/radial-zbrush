// RadialBridge.txt
// Radial menu bridge for ZBrush 2023/2024 (ZScript).
//
// Load:  ZScript > Load > this file. ZBrush compiles it to RadialBridge.zsc.
// Load always: copy RadialBridge.zsc to ZBrush's ZStartup\ZPlugs64 folder.
//
// The external app writes one line to <rbDir>command.txt, in the format:
//     press|Brush:Move
//     load|C:/caminho/Clay.ZBP
// and presses the Run button's hotkey. Run reads the line, executes it and deletes the file.

// Command folder. Must be the same as "command_dir" in config/menus.json.
// Use forward slashes (/) and end with /.
[VarDef, rbDir, "J:/CODING/radial-zbrush/zbrush_bridge/data/"]

// Load Brush button. Confirm it with the "Check" button (Phase 0).
[VarDef, rbLoadItem, "Brush:Load Brush"]

[VarDef, rbFile, ""]
[VarDef, rbLine, ""]
[VarDef, rbKind, ""]
[VarDef, rbArg, ""]
[VarDef, rbSep, 0]
[VarDef, rbSize, 0]
[VarDef, rbAsk, ""]

// Shows a quick message (1.5 s).
[RoutineDef, rbShow,
	[Note, rbMsg, , 1.5]
, rbMsg]

// Runs the command held in rbLine ("kind|argument").
[RoutineDef, rbDo,
	[VarSet, rbSep, [StrFind, "|", rbLine]]
	[If, rbSep < 1,
		[RoutineCall, rbShow, [StrMerge, "Radial: invalid command: ", rbLine]]
	,
		[VarSet, rbKind, [StrExtract, rbLine, 0, rbSep - 1]]
		[VarSet, rbArg, [StrExtract, rbLine, rbSep + 1, [StrLength, rbLine] - 1]]
		[If, [StrFind, "press", rbKind] == 0,
			[If, [IExists, rbArg],
				[IPress, rbArg]
			,
				[RoutineCall, rbShow, [StrMerge, "Radial: item does not exist: ", rbArg]]
			]
		]
		[If, [StrFind, "load", rbKind] == 0,
			[If, [FileExists, rbArg],
				[If, [IExists, rbLoadItem],
					[FileNameSetNext, rbArg]
					[IPress, rbLoadItem]
				,
					[RoutineCall, rbShow, [StrMerge, "Radial: button not found: ", rbLoadItem]]
				]
			,
				[RoutineCall, rbShow, [StrMerge, "Radial: file not found: ", rbArg]]
			]
		]
	]
]

// Reads command.txt, deletes it and runs it.
[RoutineDef, rbRun,
	[VarSet, rbFile, [StrMerge, rbDir, "command.txt"]]
	[If, [FileExists, rbFile],
		[If, [MemGetSize, rbMem], [MemDelete, rbMem]]
		[VarSet, rbSize, [MemCreateFromFile, rbMem, rbFile]]
		[VarSet, rbLine, ""]
		[If, rbSize > 0,
			[MemReadString, rbMem, rbLine, 0, 1]
			[MemDelete, rbMem]
		]
		[FileDelete, rbFile]
		[RoutineCall, rbDo]
	,
		[RoutineCall, rbShow, "Radial: no pending command"]
	]
]

[ISubPalette, "ZPlugin:Radial Bridge"]

// Main button. Assign the hotkey: Ctrl+Alt+click it and press the combination
// of "trigger_hotkey" from the config. Then save in Preferences > Hotkeys.
[IButton, "ZPlugin:Radial Bridge:Run", "Runs the radial menu command. Assign the hotkey with Ctrl+Alt+click.",
	[RoutineCall, rbRun]
, 0, 1]

// --- Phase 0 tests ---

[IButton, "ZPlugin:Radial Bridge:Test Press", "Phase 0: asks for an item path (e.g. Brush:Move) and presses it.",
	[VarSet, rbAsk, [StrAsk, "Brush:Move", "Brush item path"]]
	[If, [StrLength, rbAsk] > 0,
		[VarSet, rbLine, [StrMerge, "press|", rbAsk]]
		[RoutineCall, rbDo]
	]
, 0, 0.5]

[IButton, "ZPlugin:Radial Bridge:Test Load", "Phase 0: asks for a .ZBP path and loads it.",
	[VarSet, rbAsk, [StrAsk, "", "Full path of the .ZBP"]]
	[If, [StrLength, rbAsk] > 0,
		[VarSet, rbLine, [StrMerge, "load|", rbAsk]]
		[RoutineCall, rbDo]
	]
, 0, 0.5]

[IButton, "ZPlugin:Radial Bridge:Check", "Phase 0: checks that the Load Brush button exists.",
	[If, [IExists, rbLoadItem],
		[RoutineCall, rbShow, [StrMerge, "OK, exists: ", rbLoadItem]]
	,
		[RoutineCall, rbShow, [StrMerge, "DOES NOT exist: ", rbLoadItem]]
	]
, 0, 1]
